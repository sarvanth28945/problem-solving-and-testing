class Result {

    /*
     * Complete the 'matrixRotation' function below.
     *
     * The function accepts following parameters:
     *  1. 2D_INTEGER_ARRAY matrix
     *  2. INTEGER r
     */
    public static void matrixRotation(List<List<Integer>> matrix, int r) {
        int m = matrix.size();
        int n = matrix.get(0).size();
        
        int numLayers = Math.min(m, n) / 2;
        int[][] res = new int[m][n];
        
        for (int layer = 0; layer < numLayers; layer++) {
            List<Integer> layerElements = new ArrayList<>();
            
            int top = layer;
            int bottom = m - 1 - layer;
            int left = layer;
            int right = n - 1 - layer;
            
            // 1. Traverse Top row
            for (int j = left; j <= right; j++) {
                layerElements.add(matrix.get(top).get(j));
            }
            
            // 2. Traverse Right column
            for (int i = top + 1; i <= bottom; i++) {
                layerElements.add(matrix.get(i).get(right));
            }
            
            // 3. Traverse Bottom row
            for (int j = right - 1; j >= left; j--) {
                layerElements.add(matrix.get(bottom).get(j));
            }
            
            // 4. Traverse Left column
            for (int i = bottom - 1; i > top; i--) {
                layerElements.add(matrix.get(i).get(left));
            }
            
            int len = layerElements.size();
            int effectiveR = r % len;
            int idx = effectiveR;
            
            // Fill Top row
            for (int j = left; j <= right; j++) {
                res[top][j] = layerElements.get(idx);
                idx = (idx + 1) % len;
            }
            
            // Fill Right column
            for (int i = top + 1; i <= bottom; i++) {
                res[i][right] = layerElements.get(idx);
                idx = (idx + 1) % len;
            }
            
            // Fill Bottom row
            for (int j = right - 1; j >= left; j--) {
                res[bottom][j] = layerElements.get(idx);
                idx = (idx + 1) % len;
            }
            
            // Fill Left column
            for (int i = bottom - 1; i > top; i--) {
                res[i][left] = layerElements.get(idx);
                idx = (idx + 1) % len;
            }
        }
        
        // Print output matrix
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                sb.append(res[i][j]).append(j == n - 1 ? "" : " ");
            }
            sb.append("\n");
        }
        System.out.print(sb.toString());
    }
}

public class Solution {
    public static void main(String[] args) throws IOException {
        BufferedReader bufferedReader = new BufferedReader(new InputStreamReader(System.in));

        String[] firstMultipleInput = bufferedReader.readLine().replaceAll("\\s+$", "").split(" ");

        int m = Integer.parseInt(firstMultipleInput[0]);
        int n = Integer.parseInt(firstMultipleInput[1]);
        int r = Integer.parseInt(firstMultipleInput[2]);

        List<List<Integer>> matrix = new ArrayList<>();

        IntStream.range(0, m).forEach(i -> {
            try {
                matrix.add(
                    Stream.of(bufferedReader.readLine().replaceAll("\\s+$", "").split(" "))
                        .map(Integer::parseInt)
                        .collect(toList())
                );
            } catch (IOException ex) {
                throw new RuntimeException(ex);
            }
        });

        Result.matrixRotation(matrix, r);

        bufferedReader.close();
    }
}
