import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    t = int(data[0])
    results = []
    
    for i in range(1, t + 1):
        s = data[i]
        n = len(s)
        
        # Determine the length of each half
        half_len = n // 2
        
        # Extract the first half and second half
        left_half = s[:half_len]
        
        # If length is odd, skip the middle character for right half
        if n % 2 == 0:
            right_half = s[half_len:]
        else:
            right_half = s[half_len + 1:]
            
        # Compare sorted characters of both halves
        if sorted(left_half) == sorted(right_half):
            results.append("YES")
        else:
            results.append("NO")
            
    print('\n'.join(results))

if __name__ == '__main__':
    solve()
